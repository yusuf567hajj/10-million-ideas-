export default function App() {
  // --- STATE MANAGEMENT ---
  const [theme, setTheme] = useState('dark');
  const [activeCategory, setActiveCategory] = useState('all');
  const [searchQuery, setSearchQuery] = useState('');
  const [stories, setStories] = useState(INITIAL_STORIES);
  const [selectedStory, setSelectedStory] = useState(null);
  const [activeTab, setActiveTab] = useState('home'); // 'home', 'rss', 'ai-assistant', 'admin'
  const [vipPassActive, setVipPassActive] = useState(false);

  // Visitor & Entitlement State
  const [visitorToken, setVisitorToken] = useState('');
  const [complimentaryUsed, setComplimentaryUsed] = useState(false);
  const [purchasedStoryIds, setPurchasedStoryIds] = useState([]);
  const [transactions, setTransactions] = useState([]);
  const [toastMessage, setToastMessage] = useState(null);

  // M-Pesa Modal State
  const [showMpesaModal, setShowMpesaModal] = useState(false);
  const [mpesaPhoneNumber, setMpesaPhoneNumber] = useState('');
  const [pendingStoryToUnlock, setPendingStoryToUnlock] = useState(null);
  const [mpesaStatus, setMpesaStatus] = useState('idle'); // idle, initiating, prompt_sent, pin_entered, success, error
  const [mpesaReceipt, setMpesaReceipt] = useState('');
  const [selectedBundle, setSelectedBundle] = useState('single'); // single, bundle3, vipDaily

  // Admin & Audio state
  const [adminPin, setAdminPin] = useState('');
  const [isAdminAuthenticated, setIsAdminAuthenticated] = useState(true);
  const [isPlayingAudio, setIsPlayingAudio] = useState(false);
  const [editingStory, setEditingStory] = useState(null);

  // AI Assistant generator state
  const [aiTopic, setAiTopic] = useState('');
  const [aiGeneratedDraft, setAiGeneratedDraft] = useState(null);

  // --- COMPLIMENTARY & ENTITLEMENT INITIALIZATION ---
  useEffect(() => {
    let token = localStorage.getItem('10m_visitor_token');
    if (!token) {
      token = 'KE_VISITOR_' + Math.random().toString(36).substring(2, 10).toUpperCase();
      localStorage.setItem('10m_visitor_token', token);
    }
    setVisitorToken(token);

    const compUsed = localStorage.getItem('10m_complimentary_used') === 'true';
    setComplimentaryUsed(compUsed);

    const savedPurchases = JSON.parse(localStorage.getItem('10m_purchases') || '[]');
    setPurchasedStoryIds(savedPurchases);

    const savedTx = JSON.parse(localStorage.getItem('10m_transactions') || '[]');
    setTransactions(savedTx);
  }, []);

  const triggerToast = (msg, type = 'info') => {
    setToastMessage({ text: msg, type });
    setTimeout(() => setToastMessage(null), 4000);
  };

  // Check if story is entitled
  const isEntitled = (story) => {
    if (!story) return false;
    if (vipPassActive) return true;
    return purchasedStoryIds.includes(story.id);
  };

  // Open Story Reader logic
  const handleOpenStory = (story) => {
    setSelectedStory(story);

    // If already unlocked or VIP active, open directly
    if (isEntitled(story)) return;

    // Check Complimentary Trial
    if (!complimentaryUsed && story.isComplimentaryEligible) {
      // Grant complimentary access
      const updatedPurchases = [...purchasedStoryIds, story.id];
      setPurchasedStoryIds(updatedPurchases);
      localStorage.setItem('10m_purchases', JSON.stringify(updatedPurchases));
      setComplimentaryUsed(true);
      localStorage.setItem('10m_complimentary_used', 'true');
      triggerToast('🎉 Complimentary Full Access Unlocked! Enjoy your 1st free reading on 10 Million Ideas Kenya.', 'success');
    }
  };

  // Initiate M-Pesa Purchase Modal
  const triggerMpesaCheckout = (story) => {
    setPendingStoryToUnlock(story);
    setMpesaStatus('idle');
    setMpesaReceipt('');
    setShowMpesaModal(true);
  };

  // Simulate Daraja STK Push Backend Callbacks
  const handleProcessMpesaPayment = () => {
    if (!mpesaPhoneNumber || mpesaPhoneNumber.length < 9) {
      triggerToast('Please enter a valid Kenyan Safaricom phone number', 'error');
      return;
    }

    setMpesaStatus('initiating');

    setTimeout(() => {
      setMpesaStatus('prompt_sent');
      
      // Simulate User entering PIN on phone
      setTimeout(() => {
        setMpesaStatus('pin_entered');
        
        // Backend callback validation simulation
        setTimeout(() => {
          const receiptCode = 'QHK' + Math.floor(100000 + Math.random() * 900000);
          setMpesaReceipt(receiptCode);
          setMpesaStatus('success');

          let amountPaid = pendingStoryToUnlock ? pendingStoryToUnlock.price : 10;
          let newlyPurchased = [...purchasedStoryIds];

          if (selectedBundle === 'vipDaily') {
            setVipPassActive(true);
            amountPaid = 50;
          } else if (selectedBundle === 'bundle3') {
            amountPaid = 25;
            // Unlock current + next 2 stories
            const nextStories = stories.slice(0, 3).map(s => s.id);
            newlyPurchased = Array.from(new Set([...newlyPurchased, ...nextStories]));
          } else if (pendingStoryToUnlock) {
            newlyPurchased.push(pendingStoryToUnlock.id);
          }

          setPurchasedStoryIds(newlyPurchased);
          localStorage.setItem('10m_purchases', JSON.stringify(newlyPurchased));

          // Log transaction
          const newTx = {
            id: receiptCode,
            phone: mpesaPhoneNumber,
            amount: amountPaid,
            item: pendingStoryToUnlock ? pendingStoryToUnlock.title : 'VIP Pass',
            date: new Date().toLocaleTimeString(),
            status: 'COMPLETED'
          };
          const updatedTxs = [newTx, ...transactions];
          setTransactions(updatedTxs);
          localStorage.setItem('10m_transactions', JSON.stringify(updatedTxs));

          triggerToast(`✅ Payment Verified! Receipt ${receiptCode}. Full content unlocked.`, 'success');
        }, 1500);
      }, 2000);
    }, 1200);
  };

  // Filtered stories based on category & search
  const filteredStories = useMemo(() => {
    return stories.filter(story => {
      const matchesCat = activeCategory === 'all' || story.category === activeCategory;
      const matchesSearch = story.title.toLowerCase().includes(searchQuery.toLowerCase()) || 
                            story.teaser.toLowerCase().includes(searchQuery.toLowerCase());
      return matchesCat && matchesSearch;
    });
  }, [stories, activeCategory, searchQuery]);

  // AI Headline & Content Generator Simulation
  const handleGenerateAiDraft = () => {
    if (!aiTopic) return;
    setAiGeneratedDraft({
      headline: `The Hidden Truth About ${aiTopic} Every Kenyan Should Know`,
      teaser: `An exclusive investigation uncovering surprising facts, strategic opportunities, and actionable advice surrounding ${aiTopic}.`,
      suggestedPrice: 20,
      suggestedCategory: 'business',
      keywords: ['Kenya', aiTopic, 'Trending2026', 'Opportunity']
    });
  };

  // Header Component
  return (
    <div className={`min-h-screen transition-colors duration-200 ${theme === 'dark' ? 'bg-slate-950 text-slate-100' : 'bg-slate-50 text-slate-900'} font-sans pb-20 md:pb-8`}>
      
      {/* --- TOAST ALERT --- */}
      {toastMessage && (
        <div className="fixed top-4 right-4 z-50 max-w-md animate-bounce">
          <div className={`p-4 rounded-xl shadow-2xl flex items-center space-x-3 border ${
            toastMessage.type === 'success' 
              ? 'bg-emerald-950 border-emerald-500/50 text-emerald-200' 
              : 'bg-rose-950 border-rose-500/50 text-rose-200'
          }`}>
            <Zap className="w-5 h-5 text-amber-400 shrink-0" />
            <p className="text-sm font-medium">{toastMessage.text}</p>
          </div>
        </div>
      )}

      {/* --- TOP NAVIGATION HEADER --- */}
      <header className={`sticky top-0 z-40 border-b backdrop-blur-md ${theme === 'dark' ? 'bg-slate-950/90 border-slate-800' : 'bg-white/90 border-slate-200'}`}>
        <div className="max-w-7xl mx-auto px-4 py-3 flex items-center justify-between">
          
          {/* Logo & Tagline */}
          <div className="flex items-center space-x-3 cursor-pointer" onClick={() => { setSelectedStory(null); setActiveTab('home'); }}>
            <div className="w-10 h-10 rounded-xl bg-gradient-to-tr from-emerald-600 via-amber-500 to-emerald-400 flex items-center justify-center font-extrabold text-slate-950 shadow-lg shadow-emerald-900/30">
              10M
            </div>
            <div>
              <h1 className="font-black text-lg md:text-xl tracking-tight leading-none bg-gradient-to-r from-emerald-400 via-amber-300 to-emerald-200 bg-clip-text text-transparent">
                10 MILLION IDEAS
              </h1>
              <span className="text-[10px] tracking-widest uppercase font-semibold text-amber-500 block">
                KENYA • DISCOVER MORE
              </span>
            </div>
          </div>

          {/* Search Bar (Desktop) */}
          <div className="hidden md:flex flex-1 max-w-md mx-8 relative">
            <Search className="w-4 h-4 absolute left-3 top-3 text-slate-400" />
            <input 
              type="text"
              placeholder="Search EPL, news, business ideas, recipes..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className={`w-full pl-9 pr-4 py-2 text-sm rounded-full border focus:outline-none focus:ring-2 focus:ring-emerald-500 ${
                theme === 'dark' ? 'bg-slate-900 border-slate-800 text-slate-100' : 'bg-slate-100 border-slate-200 text-slate-900'
              }`}
            />
          </div>

          {/* User Status / Entitlement Badge & Controls */}
          <div className="flex items-center space-x-3">
            {/* Complimentary Status Badge */}
            <div className={`hidden sm:flex items-center space-x-1.5 px-3 py-1 rounded-full text-xs font-semibold border ${
              complimentaryUsed 
                ? 'bg-amber-500/10 text-amber-400 border-amber-500/30' 
                : 'bg-emerald-500/10 text-emerald-400 border-emerald-500/30 animate-pulse'
            }`}>
              {complimentaryUsed ? <Lock className="w-3.5 h-3.5" /> : <Sparkles className="w-3.5 h-3.5" />}
              <span>{complimentaryUsed ? '1 Free Pass Used' : '1 Free Pass Available'}</span>
            </div>

            {/* Currency & VIP Pass status */}
            <button 
              onClick={() => triggerMpesaCheckout(null)}
              className="px-3 py-1.5 rounded-lg bg-emerald-600 hover:bg-emerald-500 text-white font-semibold text-xs flex items-center space-x-1 shadow-md transition"
            >
              <Smartphone className="w-3.5 h-3.5" />
              <span>{vipPassActive ? 'VIP Active' : 'Get VIP Pass'}</span>
            </button>

            {/* Dark/Light Mode Toggle */}
            <button 
              onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
              className={`p-2 rounded-lg border ${theme === 'dark' ? 'border-slate-800 bg-slate-900 text-amber-400' : 'border-slate-200 bg-slate-100 text-slate-700'}`}
              title="Toggle Theme"
            >
              {theme === 'dark' ? <Sun className="w-4 h-4" /> : <Moon className="w-4 h-4" />}
            </button>
          </div>
        </div>

        {/* Mobile Search Bar */}
        <div className="md:hidden px-4 pb-2">
          <div className="relative">
            <Search className="w-4 h-4 absolute left-3 top-2.5 text-slate-400" />
            <input 
              type="text"
              placeholder="Search stories, sports, money..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className={`w-full pl-9 pr-4 py-1.5 text-xs rounded-full border ${
                theme === 'dark' ? 'bg-slate-900 border-slate-800 text-slate-100' : 'bg-slate-100 border-slate-200 text-slate-900'
              }`}
            />
          </div>
        </div>
      </header>

      {/* --- SUB NAVIGATION TAB BAR (DESKTOP/MAIN) --- */}
      <div className={`border-b ${theme === 'dark' ? 'bg-slate-900/50 border-slate-800' : 'bg-slate-100/50 border-slate-200'} sticky top-14 z-30 backdrop-blur-sm`}>
        <div className="max-w-7xl mx-auto px-4 flex items-center justify-between overflow-x-auto no-scrollbar py-2">
          <div className="flex space-x-1">
            <button 
              onClick={() => { setActiveTab('home'); setSelectedStory(null); }}
              className={`px-3 py-1.5 rounded-lg text-xs font-semibold whitespace-nowrap flex items-center space-x-1.5 transition ${
                activeTab === 'home' 
                  ? 'bg-emerald-500 text-slate-950 font-bold' 
                  : 'text-slate-400 hover:text-slate-100'
              }`}
            >
              <Sparkles className="w-3.5 h-3.5" />
              <span>Editorial Feed</span>
            </button>

            <button 
              onClick={() => setActiveTab('rss')}
              className={`px-3 py-1.5 rounded-lg text-xs font-semibold whitespace-nowrap flex items-center space-x-1.5 transition ${
                activeTab === 'rss' 
                  ? 'bg-emerald-500 text-slate-950 font-bold' 
                  : 'text-slate-400 hover:text-slate-100'
              }`}
            >
              <Radio className="w-3.5 h-3.5" />
              <span>Live RSS Trends</span>
            </button>

            <button 
              onClick={() => setActiveTab('ai-assistant')}
              className={`px-3 py-1.5 rounded-lg text-xs font-semibold whitespace-nowrap flex items-center space-x-1.5 transition ${
                activeTab === 'ai-assistant' 
                  ? 'bg-emerald-500 text-slate-950 font-bold' 
                  : 'text-slate-400 hover:text-slate-100'
              }`}
            >
              <Zap className="w-3.5 h-3.5" />
              <span>AI Newsroom Drafts</span>
            </button>

            <button 
              onClick={() => setActiveTab('admin')}
              className={`px-3 py-1.5 rounded-lg text-xs font-semibold whitespace-nowrap flex items-center space-x-1.5 transition ${
                activeTab === 'admin' 
                  ? 'bg-amber-500 text-slate-950 font-bold' 
                  : 'text-slate-400 hover:text-slate-100'
              }`}
            >
              <Sliders className="w-3.5 h-3.5" />
              <span>Publisher Dashboard</span>
            </button>
          </div>

          <div className="hidden lg:flex items-center space-x-2 text-xs text-slate-400">
            <span className="w-2 h-2 rounded-full bg-emerald-500 animate-pulse"></span>
            <span>Live Daraja STK Gate Connected</span>
          </div>
        </div>
      </div>

      {/* --- MAIN CONTENT SWITCHER --- */}
      <main className="max-w-7xl mx-auto px-4 py-6">

        {/* 1. INDIVIDUAL STORY READER VIEW */}
        {selectedStory ? (
          <div className="max-w-4xl mx-auto space-y-6">
            <button 
              onClick={() => setSelectedStory(null)}
              className="text-xs font-bold text-emerald-400 flex items-center space-x-1 hover:underline"
            >
              ← Back to Editorial Discovery
            </button>

            <article className={`rounded-2xl border overflow-hidden ${theme === 'dark' ? 'bg-slate-900 border-slate-800' : 'bg-white border-slate-200'} shadow-2xl`}>
              {/* Cover Banner */}
              <div className="relative h-72 md:h-96 w-full overflow-hidden">
                <img 
                  src={selectedStory.thumbnail} 
                  alt={selectedStory.title} 
                  className="w-full h-full object-cover"
                />
                <div className="absolute inset-0 bg-gradient-to-t from-slate-950 via-slate-950/40 to-transparent"></div>
                <div className="absolute bottom-6 left-6 right-6 space-y-2">
                  <div className="flex items-center space-x-2">
                    <span className="px-3 py-1 rounded-full text-xs font-bold bg-emerald-500 text-slate-950 uppercase tracking-wide">
                      {selectedStory.category}
                    </span>
                    <span className="px-2.5 py-0.5 rounded-full text-xs bg-slate-900/80 text-slate-300 backdrop-blur-sm">
                      {selectedStory.readTime}
                    </span>
                  </div>
                  <h1 className="text-2xl md:text-4xl font-extrabold text-white leading-tight">
                    {selectedStory.title}
                  </h1>
                </div>
              </div>

              {/* Story Header Details */}
              <div className="p-6 md:p-8 space-y-6 border-b border-slate-800">
                <div className="flex items-center justify-between text-xs text-slate-400">
                  <div className="flex items-center space-x-3">
                    <div className="w-8 h-8 rounded-full bg-emerald-600 flex items-center justify-center text-white font-bold text-xs">
                      {selectedStory.author[0]}
                    </div>
                    <div>

            <div className="p-6 rounded-2xl bg-slate-900 border border-slate-800 space-y-4">
              <h3 className="font-bold text-white text-sm flex items-center space-x-2">
                <Key className="w-4 h-4 text-emerald-400" />
                <span>Safaricom Daraja API Gateway Configuration</span>

              <div className="grid grid-cols-1 md:grid-cols-3 gap-3 text-xs">
                <div>
                  <span className="text-slate-500 block">CONSUMER_KEY</span>
                  <code className="bg-slate-950 px-2 py-1 rounded text-emerald-400 block mt-1 overflow-x-auto">
                    MPESA_PROD_998127361
                  </code>
                </div>
                <div>
                  <span className="text-slate-500 block">SHORTCODE</span>
                  <code className="bg-slate-950 px-2 py-1 rounded text-emerald-400 block mt-1">
                    174379 (Paybill/STK)
                  </code>
                </div>
                <div>
                  <span className="text-slate-500 block">CALLBACK_ENDPOINT</span>
                  <code className="bg-slate-950 px-2 py-1 rounded text-emerald-400 block mt-1 truncate">
                    /api/payments/mpesa/callback
                  </code>
                </div>
              </div>
            </div>

            {/* M-PESA TRANSACTIONS LEDGER */}
            <div className="p-6 rounded-2xl bg-slate-900 border border-slate-800 space-y-4 overflow-x-auto">
              <h3 className="font-bold text-white text-sm">Recent M-Pesa STK Transactions</h3>
              {transactions.length === 0 ? (
                <p className="text-xs text-slate-500">No completed payments recorded yet in this session.</p>
              ) : (
                <table className="w-full text-left text-xs text-slate-300">
                  <thead className="bg-slate-950 text-slate-400 uppercase text-[10px]">
                    <tr>
                      <th className="p-3">Receipt Code</th>
                      <th className="p-3">Phone Number</th>
                      <th className="p-3">Amount</th>
                      <th className="p-3">Purchased Content</th>
                      <th className="p-3">Time</th>
                      <th className="p-3">Status</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-800">
                    {transactions.map(tx => (
                      <tr key={tx.id} className="hover:bg-slate-800/50">
                        <td className="p-3 font-mono text-amber-400">{tx.id}</td>
                        <td className="p-3">{tx.phone}</td>
                        <td className="p-3 font-bold text-emerald-400">KSh {tx.amount}</td>
                        <td className="p-3 max-w-xs truncate">{tx.item}</td>

                          <span className="px-2 py-0.5 rounded text-[10px] font-bold bg-emerald-500/20 text-emerald-400 border border-emerald-500/30">
                            {tx.status}
                          </span>
                        </td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              )}
            </div>
          </div>

        ) : null}

      </main>

      {/* --- M-PESA STK PUSH CHECKOUT MODAL --- */}
      {showMpesaModal && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/80 backdrop-blur-md animate-fadeIn">
          <div className="w-full max-w-md rounded-3xl bg-slate-900 border-2 border-emerald-500/40 p-6 md:p-8 space-y-6 shadow-2xl relative">
            <button 
              onClick={() => setShowMpesaModal(false)}
              className="absolute top-4 right-4 p-2 rounded-full text-slate-400 hover:text-white bg-slate-800"
            >
              <X className="w-4 h-4" />
            </button>

            {/* Header */}
            <div className="text-center space-y-2">
              <div className="w-12 h-12 rounded-2xl bg-emerald-500 text-slate-950 flex items-center justify-center mx-auto shadow-lg shadow-emerald-950">
                <Smartphone className="w-6 h-6" />
              </div>
              <h3 className="text-xl font-extrabold text-white">M-Pesa Express Checkout</h3>
              <p className="text-xs text-slate-400">
                Safaricom STK Push instant unlock for 10 Million Ideas Kenya
              </p>
            </div>

            {/* Select Options */}
            <div className="space-y-2">
              <label className="block text-xs font-bold text-slate-300">Choose Purchase Option:</label>
              <div className="grid grid-cols-3 gap-2">
                <button
                  onClick={() => setSelectedBundle('single')}
                  className={`p-3 rounded-xl border text-center transition ${
                    selectedBundle === 'single'
                      ? 'border-emerald-500 bg-emerald-500/10 text-emerald-300'
                      : 'border-slate-800 bg-slate-950 text-slate-400'
                  }`}
                >
                  <span className="block text-xs font-bold">This Story</span>
                  <span className="text-sm font-extrabold text-white">KSh {pendingStoryToUnlock ? pendingStoryToUnlock.price : 10}</span>
                </button>

                <button
                  onClick={() => setSelectedBundle('bundle3')}
                  className={`p-3 rounded-xl border text-center transition ${
                    selectedBundle === 'bundle3'
                      ? 'border-emerald-500 bg-emerald-500/10 text-emerald-300'
                      : 'border-slate-800 bg-slate-950 text-slate-400'
                  }`}
                >
                  <span className="block text-xs font-bold">3 Stories</span>
                  <span className="text-sm font-extrabold text-amber-400">KSh 25</span>
                </button>

                <button
                  onClick={() => setSelectedBundle('vipDaily')}
                  className={`p-3 rounded-xl border text-center transition ${
                    selectedBundle === 'vipDaily'
                      ? 'border-emerald-500 bg-emerald-500/10 text-emerald-300'
                      : 'border-slate-800 bg-slate-950 text-slate-400'
                  }`}
                >
                  <span className="block text-xs font-bold">VIP 24H Pass</span>
                  <span className="text-sm font-extrabold text-emerald-400">KSh 50</span>
                </button>
              </div>
            </div>

            {/* M-Pesa Phone Input & Process */}
            {mpesaStatus === 'idle' && (
              <div className="space-y-4">
                <div>
                  <label className="block text-xs text-slate-400 mb-1 font-semibold">
                    Enter Safaricom Phone Number:
                  </label>
                  <div className="relative">
                    <span className="absolute left-3 top-3 text-xs font-bold text-slate-500">+254</span>
                    <input 
                      type="text" 
                      placeholder="712345678 or 0712345678"
                      value={mpesaPhoneNumber}
                      onChange={(e) => setMpesaPhoneNumber(e.target.value)}
                      className="w-full pl-14 pr-4 py-2.5 rounded-xl bg-slate-950 border border-slate-800 text-white text-sm focus:outline-none focus:ring-2 focus:ring-emerald-500"
                    />
                  </div>
                </div>

                <button 
                  onClick={handleProcessMpesaPayment}
                  className="w-full py-3.5 rounded-xl bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-extrabold text-sm shadow-xl flex items-center justify-center space-x-2 transition"
                >
                  <span>Request M-Pesa STK Push</span>
                  <ChevronRight className="w-4 h-4" />
                </button>
              </div>
            )}

            {/* STK Push Lifecycle Loader & Prompts */}
            {mpesaStatus !== 'idle' && (
              <div className="py-6 text-center space-y-4">
                {mpesaStatus === 'initiating' && (
                  <div className="space-y-3">
                    <RefreshCw className="w-8 h-8 text-emerald-400 animate-spin mx-auto" />
                    <p className="text-xs text-slate-300 font-semibold">Connecting to Safaricom Daraja API Gateway...</p>
                  </div>
                )}

                {mpesaStatus === 'prompt_sent' && (
                  <div className="space-y-3 animate-pulse">
                    <div className="w-12 h-12 rounded-full bg-amber-500/20 text-amber-400 flex items-center justify-center mx-auto border border-amber-500/40">
                      <Smartphone className="w-6 h-6" />
                    </div>
                    <p className="text-sm font-bold text-white">STK Push Sent to {mpesaPhoneNumber}</p>
                    <p className="text-xs text-slate-400">Please check your mobile phone screen and enter your M-Pesa PIN.</p>
                  </div>
                )}

                {mpesaStatus === 'pin_entered' && (
                  <div className="space-y-3">
                    <RefreshCw className="w-8 h-8 text-amber-400 animate-spin mx-auto" />
                    <p className="text-xs text-amber-300 font-semibold">Validating Transaction Callback from Safaricom...</p>
                  </div>
                )}

                {mpesaStatus === 'success' && (
                  <div className="space-y-4 animate-fadeIn">
                    <div className="w-12 h-12 rounded-full bg-emerald-500 text-slate-950 flex items-center justify-center mx-auto shadow-lg">
                      <Check className="w-6 h-6 stroke-[3]" />
                    </div>
                    <div>
                      <h4 className="text-lg font-black text-white">Payment Confirmed!</h4>
                      <p className="text-xs text-slate-400 mt-1">Receipt Code: <span className="font-mono text-amber-400 font-bold">{mpesaReceipt}</span></p>
                    </div>

                    <button 
                      onClick={() => setShowMpesaModal(false)}
                      className="w-full py-3 rounded-xl bg-emerald-500 text-slate-950 font-bold text-xs"
                    >
                      Return & Enjoy Story
                    </button>
                  </div>
                )}
              </div>
            )}
          </div>
        </div>
      )}

      {/* --- FOOTER COMPLIANCE & LEGAL --- */}
      <footer className={`mt-16 border-t ${theme === 'dark' ? 'bg-slate-950 border-slate-900 text-slate-400' : 'bg-slate-100 border-slate-200 text-slate-600'} py-12 px-4`}>
        <div className="max-w-7xl mx-auto grid grid-cols-1 md:grid-cols-4 gap-8 text-xs">
          <div className="space-y-3">
            <div className="flex items-center space-x-2">
              <div className="w-7 h-7 rounded-lg bg-emerald-500 flex items-center justify-center font-bold text-slate-950 text-xs">
                10M
              </div>
              <span className="font-extrabold text-slate-100 text-sm tracking-tight">10 MILLION IDEAS KENYA</span>
            </div>
            <p className="text-slate-500 leading-relaxed">
              Discover More. Know More. Enjoy More. Kenya's premier anonymous paid digital editorial and breaking story platform.
            </p>
          </div>

          <div>
            <h4 className="font-bold text-slate-200 uppercase tracking-wider mb-3 text-[11px]">Editorial Categories</h4>
            <ul className="space-y-2">
              <li>EPL & Harambee Stars Football</li>
              <li>Kenya Breaking News & Governance</li>
              <li>Household Life Hacks & Budgeting</li>
              <li>Kenyan Side Hustles & Business Ideas</li>
            </ul>
          </div>

          <div>
            <h4 className="font-bold text-slate-200 uppercase tracking-wider mb-3 text-[11px]">Payment & Rights</h4>
            <ul className="space-y-2">
              <li>Safaricom M-Pesa STK Express</li>
              <li>Complimentary Entitlement Token Policy</li>
              <li>Privacy & Kenyan Data Protection Compliance</li>
              <li>Instant M-Pesa Refund Protocols</li>
            </ul>
          </div>

          <div>
            <h4 className="font-bold text-slate-200 uppercase tracking-wider mb-3 text-[11px]">Publisher & Sources</h4>
            <p className="text-slate-500 leading-relaxed">
              Curated stories synthesized from authorized partner wire feeds and verified local editorial contributors across Kenya.
            </p>
            <p className="mt-3 text-[10px] text-slate-600">
              © 2026 10 Million Ideas Kenya. All Rights Reserved.
            </p>
          </div>
        </div>
      </footer>

      {/* --- MOBILE STICKY BOTTOM NAVIGATION BAR --- */}
      <nav className={`md:hidden fixed bottom-0 inset-x-0 z-40 border-t ${theme === 'dark' ? 'bg-slate-950/95 border-slate-800 text-slate-400' : 'bg-white/95 border-slate-200 text-slate-600'} backdrop-blur-lg flex justify-around py-2`}>
        <button 
          onClick={() => { setSelectedStory(null); setActiveTab('home'); }}
          className={`flex flex-col items-center space-y-1 ${activeTab === 'home' && !selectedStory ? 'text-emerald-400 font-bold' : ''}`}
        >
          <Sparkles className="w-5 h-5" />
          <span className="text-[10px]">Home</span>
        </button>

        <button 
          onClick={() => { setSelectedStory(null); setActiveCategory('football'); setActiveTab('home'); }}
          className={`flex flex-col items-center space-y-1 ${activeCategory === 'football' ? 'text-emerald-400 font-bold' : ''}`}
        >
          <Trophy className="w-5 h-5" />
          <span className="text-[10px]">Football</span>
        </button>

        <button 
          onClick={() => setActiveTab('rss')}
          className={`flex flex-col items-center space-y-1 ${activeTab === 'rss' ? 'text-emerald-400 font-bold' : ''}`}
        >
          <Radio className="w-5 h-5" />
          <span className="text-[10px]">Live RSS</span>
        </button>

        <button 
          onClick={() => triggerMpesaCheckout(null)}
          className="flex flex-col items-center space-y-1 text-amber-400 font-bold"
        >
          <Smartphone className="w-5 h-5" />
          <span className="text-[10px]">VIP Pass</span>
        </button>

        <button 
          onClick={() => setActiveTab('admin')}
          className={`flex flex-col items-center space-y-1 ${activeTab === 'admin' ? 'text-emerald-400 font-bold' : ''}`}
        >
          <Sliders className="w-5 h-5" />
          <span className="text-[10px]">Admin</span>
        </button>
      </nav>

    </div>
  );
